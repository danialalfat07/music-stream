    function normalizeSong(s) {
      if (!s) return s;
      return { 
        ...s, 
        title: displayTitle(s.title),
        duration: s.duration ? normalizeDuration(s.duration) : s.duration
      };
    }

F2. PERBAIKI PUSH HISTORY (R1 & R2)
    File: public/app.js (fungsi Library.pushHistory, sekitar baris 708).
    Lakukan perbaikan alur:
    
    1. Sebelum membersihkan subtitle, coba ambil durasi dari 
       `song.subtitle` menggunakan `durationFromSubtitle(song.subtitle)`.
    2. Simpan durasi yang didapat (dari `song.duration` atau dari 
       subtitle) ke dalam variabel `finalDuration`.
    3. Simpan durasi tersebut ke `Library.stats[song.videoId].duration` 
       agar fallback di `trackRowHTML` bisa membaca ini di masa depan.
    4. Saat membersihkan subtitle (`splitSubtitle`), pastikan hanya 
       membersihkan jika `finalDuration` benar-benar valid (tidak kosong). 
       Jika `finalDuration` kosong, JANGAN buang teks dari subtitle.
    5. Simpan `finalDuration` ke dalam record history `rec.duration`.

    Contoh logika:
    let finalDuration = song.duration || durationFromSubtitle(song.subtitle);
    if (finalDuration) {
        // 1. Simpan ke stats
        if (!this.stats[song.videoId]) this.stats[song.videoId] = {};
        this.stats[song.videoId].duration = finalDuration;
        // 2. Set ke rec
        rec.duration = finalDuration;
        // 3. Baru bersihkan subtitle
        if (rec.subtitle) {
           const { rest, plays } = splitSubtitle(rec.subtitle);
           rec.subtitle = [rest, plays].filter(Boolean).join(' • ');
        }
    }

F3. UPDATE FALLBACK TAMPILAN (trackRowHTML)
    File: public/app.js (sekitar baris 4257).
    Pastikan `trackRowHTML` memprioritaskan durasi dari `Library.stats` 
    jika `it.duration` kosong dan `it.subtitle` tidak punya durasi.
    
    let secs = parseDurationSec(raw);
    if (!(secs >= 1 && secs <= 43200)) secs = durationFromSubtitle(it.subtitle);
    if (!(secs >= 1 && secs <= 43200) && it.videoId) {
        try {
            const st = Library.stats[it.videoId];
            if (st && st.duration) secs = parseDurationSec(st.duration);
        } catch {}
    }
    // Pastikan blok try-catch di atas sudah ada dan berfungsi.
