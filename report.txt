3. BUKTI A2 — migrateLegacyCaches()
--------------------------------------------------------------------
File: SongCache.java (line 195)

    public static synchronized void migrateLegacyCaches(Context c) {
        if (c == null) return;
        int moved = 0;
        try {
            File[] legacyDirs = {
                new File(c.getCacheDir(), "visionos_cache"),      // COMPLETE+meta v<=3.2.3
                new File(c.getFilesDir(), "incomplete_cache"),    // INCOMPLETE v3.2.3
                new File(c.getFilesDir(), "visionos_cache")       // jaring pengaman
            };
            File complete = completeDir(c);
            File incomplete = incompleteDir(c);
            File meta = metaDir(c);
            for (File legacy : legacyDirs) {
                if (legacy == null || !legacy.isDirectory()) continue;
                File[] files = legacy.listFiles();
                if (files == null) continue;
                for (File f : files) {
                    if (f == null || !f.isFile()) continue;
                    String name = f.getName();
                    File dst;
                    if (name.endsWith(".webm.part"))            dst = new File(incomplete, name);
                    else if (name.endsWith(".webm"))            dst = new File(complete, name);
                    else if (name.endsWith(".song.json")
                          || name.endsWith(".art")
                          || name.endsWith(".lyrics.json")
                          || name.endsWith(".lyrics.json.tmp")) dst = new File(meta, name);
                    else continue;                        // tipe tak dikenal: biarkan
                    if (dst.exists()) continue;           // idempotent
                    try { if (moveFile(f, dst)) moved++; } catch (Exception ignored) {}
                }
            }
            Log.d(TAG, "[MIGRATE] legacy caches moved=" + moved);
        } catch (Exception e) {
            Log.d(TAG, "[MIGRATE] error: " + e);
        }
    }

Sifat: sekali jalan, idempoten (skip bila file tujuan sudah ada),
memakai rename-first + fallback copy+delete, TIDAK menghapus data bila
gagal (file dibiarkan di lokasi lama). Dipanggil di
MainActivity.onCreate() (line ~102) tepat SEBELUM enforceLimit:

    try {
        SongCache.migrateLegacyCaches(this);
    } catch (Exception e) {
        Log.d(TAG_DIAG, "migrateLegacyCaches onCreate failed " + e);
    }

    }
